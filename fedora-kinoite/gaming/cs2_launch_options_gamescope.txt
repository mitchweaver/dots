# gamescope makes the stretched res look less ass on linux
# can't get it to look as good as on windows but this is much better than native
#

try:
gamescope -w 1280 -h 960 -S stretch -F nearest -f --force-grab-cursor -r 600 --framerate-limit 600 -- %command% -allow_third_party_software -w 1280 -h 960 -nojoy -novid

or:
gamescope -w 1280 -h 960 -S stretch -e -f --force-grab-cursor -r 600 --framerate-limit 600 -- %command% -allow_third_party_software -w 1280 -h 960 -nojoy -novid
